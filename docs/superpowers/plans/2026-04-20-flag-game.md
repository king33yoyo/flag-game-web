### Task 4: 地图投影函数

**Files:**

- [ ] **Step 1: 实现 Natural Earth 投影函数**

```javascript
// Natural Earth 投影（轻量内联实现）
function naturalEarthProject(lambda, phi) {
  // lambda: 经度（弧度），phi: 纬度（弧度）
  const phi2 = phi * phi;
  const phi4 = phi2 * phi2;
  const x = lambda * (0.8707 - 0.131979 * phi2 - 0.013791 * phi4);
  const y = phi * (1.007226 + 0.015085 * phi2 - 0.044475 * phi4);
  return [x, y];
}

function lonLatToXY(lon, lat, width, height) {
  const lambda = lon * Math.PI / 180;
  const phi = lat * Math.PI / 180;
  const [x, y] = naturalEarthProject(lambda, phi);
  // 归一化到 SVG 坐标：Natural Earth 投影范围约 [-2.7, 2.7] x [-1.4, 1.4]
  const scale = Math.min(width / 5.6, height / 3.0);
  const cx = width / 2;
  const cy = height / 2;
  return [cx + x * scale, cy - y * scale];
}

function geoJsonToSvgPath(geojson, width, height) {
  const coords = geojson.coordinates[0];
  const points = coords.map(([lon, lat]) => {
    const [x, y] = lonLatToXY(lon, lat, width, height);
    return `${x.toFixed(2)},${y.toFixed(2)}`;
  });
  return `M${points.join('L')}Z`;
}
```

function initGame() {
  const allCountries = COUNTRY_DATA[gameState.continent];
  // 随机选取 10 个国家（如果大洲国家数 < 10 则全选）
  const count = Math.min(10, allCountries.length);
  gameState.countries = shuffleArray([...allCountries]).slice(0, count);
  gameState.score = 0;
  gameState.errors = [];
  gameState.isComplete = false;
  gameState.startTime = Date.now();

  const c = CONTINENTS[gameState.continent];
  document.getElementById('game-region').textContent = c.name + '挑战';
  document.getElementById('score-display').textContent = `0 / ${count}`;

  startTimer();
  renderMap(gameState.continent);
  renderSlotMarkers();
  renderTray();
}

function shuffleArray(arr) {
  for (let i = arr.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
  return arr;
}

function onTouchStart(e) {

  document.addEventListener('touchend', onTouchEnd);
}

function onTouchMove(e) {
  e.preventDefault();
  if (!touchClone) return;
  const touch = e.touches[0];
  touchClone.style.left = (touch.clientX - 40) + 'px';
  touchClone.style.top = (touch.clientY - 20) + 'px';

  // 高亮悬停的国家
  const elem = document.elementFromPoint(touch.clientX, touch.clientY);
  document.querySelectorAll('.country-path').forEach(p => {
    if (!p.classList.contains('filled')) p.style.fill = '#C7D2FE';
  });
  if (elem && elem.classList.contains('country-path')) {
    elem.style.fill = '#A5B4FC';
  }
}

function onTouchEnd(e) {
  document.removeEventListener('touchmove', onTouchMove);
  document.removeEventListener('touchend', onTouchEnd);

  if (!touchClone || !touchCountryId) return;
  const touch = e.changedTouches[0];
  touchClone.remove();
  touchClone = null;

  const elem = document.elementFromPoint(touch.clientX, touch.clientY);
  const targetPath = elem && elem.closest ? elem.closest('.country-path') : null;
  const targetId = targetPath ? targetPath.dataset.id : null;

  if (targetId && targetId !== touchCountryId) {
    const chip = document.querySelector(`.flag-chip[data-id="${touchCountryId}"]`);
    if (!chip || chip.classList.contains('placed')) return;
    if (touchCountryId === targetId) {
      placeFlag(touchCountryId, targetId, true);
    } else {
      wrongDrop(touchCountryId, targetId);
    }
  } else {
    // 弹回：恢复 chip 透明度
    const chip = document.querySelector(`.flag-chip[data-id="${touchCountryId}"]`);
    if (chip) chip.style.opacity = '1';
  }

  touchCountryId = null;
}

function startTimer() {
  stopTimer();
  gameState.startTime = Date.now();
  gameState.timerInterval = setInterval(() => {
    const elapsed = Math.floor((Date.now() - gameState.startTime) / 1000);
    const min = String(Math.floor(elapsed / 60)).padStart(2, '0');
    const sec = String(elapsed % 60).padStart(2, '0');
    document.getElementById('timer-display').textContent = `${min}:${sec}`;
  }, 1000);
}

function stopTimer() {
  if (gameState.timerInterval) {
    clearInterval(gameState.timerInterval);
    gameState.timerInterval = null;
  }
}

function getElapsedTime() {
  if (!gameState.startTime) return '00:00';
  const elapsed = Math.floor((Date.now() - gameState.startTime) / 1000);
  const min = String(Math.floor(elapsed / 60)).padStart(2, '0');
  const sec = String(elapsed % 60).padStart(2, '0');
  return `${min}:${sec}`;
}

let audioCtx = null;

function getAudioCtx() {
  if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
  return audioCtx;
}

function playSound(type) {
  try {
    const ctx = getAudioCtx();
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.connect(gain);
    gain.connect(ctx.destination);

    if (type === 'correct') {
      osc.frequency.value = 800;
      gain.gain.setValueAtTime(0.3, ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.15);
      osc.start(ctx.currentTime);
      osc.stop(ctx.currentTime + 0.15);
    } else if (type === 'wrong') {
      osc.frequency.value = 300;
      osc.type = 'sawtooth';
      gain.gain.setValueAtTime(0.2, ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.2);
      osc.start(ctx.currentTime);
      osc.stop(ctx.currentTime + 0.2);
    }
  } catch (e) { /* 静默失败 */ }
}
